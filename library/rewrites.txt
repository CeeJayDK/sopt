# sopt rewrite library: verified rewrites for the program, people and AI.
#
# One rule per line:   pattern -> replacement   where condition, condition, ...   # comment
#
# Names in the pattern are variables that stand for any subexpression (the same name twice
# means the same subexpression); numbers must match exactly; functions and operators are
# ReShade FX / HLSL. Conditions (optional):
#   x const          x is a constant (or a preprocessor constant)
#   x in [lo, hi]    x's values lie in [lo, hi]; ( ) for open ends
#   x >= c           also x > c, x <= c, x < c, x != c
#   x : float3       x's type for the rule check (default float)
# Every rule is checked by `sopt --check-library` and the tests: over the variables'
# domains (default [-100, 100], narrowed by the conditions) the replacement must stay
# within rel 1e-6 of the pattern or be at least as close to exact math, under every GPU
# semantic profile. sopt --library applies the rules before the search; every rewritten
# form is verified again against the region's own budget.

# lerp(a, b, t) = a + t * (b - a)
lerp(a, b, t) -> mad(t, b - a, a)
lerp(a, b, t) -> mad(t, b, mad(-t, a, a))   where t in [0, 1]   # no b - a
1.0 - lerp(1.0, b, t) -> mad(-t, b, t)   where t in [0, 1]   # = t * (1 - b)
lerp(a, b, step(e, x)) -> x >= e ? b : a

# step(e, x) = x >= e ? 1 : 0
step(e, x) -> x >= e ? 1.0 : 0.0
a * step(e, x) -> x >= e ? a : 0.0
1.0 - step(e, x) -> x < e ? 1.0 : 0.0

# pow(x, y) = exp2(y * log2(x)): three instructions, two of them transcendental
pow(x, 1.0) -> x
pow(x, 2.0) -> x * x
pow(abs(x), 2.0) -> x * x
pow(x, 3.0) -> x * x * x
pow(abs(x), 3.0) -> abs(x) * (x * x)
pow(x, 4.0) -> (x * x) * (x * x)
pow(abs(x), 4.0) -> (x * x) * (x * x)
pow(x, 0.5) -> sqrt(x)
pow(x, 1.5) -> x * sqrt(x)
pow(x, -0.5) -> rsqrt(x)
pow(x, -1.0) -> rcp(x)

# exp, log and pow are exp2 / log2 with a multiply (DXC: exp(x) = exp2(x * log2 e), log(x) =
# log2(x) * ln 2, pow(x, y) = exp2(y * log2 x); GPUs only have the base-2 instructions), so a
# constant factor folds into that multiply. The ranges keep the exponent small enough for
# rel 1e-6 against the CPU's accurate exp / pow.
exp(x * c) -> exp2(x * (c * 1.4426950408889634))   where c const, x in [-8, 8], c in [-1, 1]
exp(x * c) -> exp2(x * (c * 1.4426950408889634))   where c const, x in [-100, 100], c in [-0.08, 0.08]
exp(x / c) -> exp2(x * (1.4426950408889634 / c))   where c const, x in [-8, 8], c in [1, 100]
log(x) * c -> log2(x) * (c * 0.6931471805599453)   where c const, x > 0
log(x) / c -> log2(x) * (0.6931471805599453 / c)   where c const, x > 0, c != 0
exp(c * log(x)) -> exp2(c * log2(x))   where c const, x > 0, c in [-8, 8]   # pow(x, c)
pow(2.0, x) -> exp2(x)
pow(c, x) -> exp2(x * log2(c))   where c const, c in [0.5, 2], x in [-8, 8]
exp2(x) * exp2(y) -> exp2(x + y)   where x in [-8, 8], y in [-8, 8]
exp(x) * exp(y) -> exp(x + y)      where x in [-8, 8], y in [-8, 8]
log2(x) + log2(y) -> log2(x * y)   where x in [1e-18, 1e18], y in [1e-18, 1e18]
log(x) + log(y) -> log(x * y)      where x in [1e-18, 1e18], y in [1e-18, 1e18]

# smoothstep(a, b, x) = s * (s * (3 - 2s)) with s = saturate((x - a) / (b - a)) (DXC)
smoothstep(a, b, x) -> saturate((x - a) / (b - a)) * (saturate((x - a) / (b - a)) * mad(saturate((x - a) / (b - a)), -2.0, 3.0))
smoothstep(0.0, 1.0, x) -> saturate(x) * (saturate(x) * mad(saturate(x), -2.0, 3.0))
smoothstep(0.0, 1.0, x) -> x * (x * mad(x, -2.0, 3.0))   where x in [0, 1]

# clamping
saturate(saturate(x)) -> saturate(x)
min(max(x, 0.0), 1.0) -> saturate(x)
max(min(x, 1.0), 0.0) -> saturate(x)
clamp(x, 0.0, 1.0) -> saturate(x)
saturate(x) -> x   where x in [0, 1]
max(x, 0.0) -> x   where x >= 0
min(x, 1.0) -> x   where x <= 1
abs(x) -> x        where x >= 0

# roots and reciprocals
abs(x) * abs(x) -> x * x
sqrt(x * x) -> abs(x)
1.0 / sqrt(x) -> rsqrt(x)
1.0 / x -> rcp(x)
x / c -> x * (1.0 / c)   where c const, c != 0

# constants merged (spirv-opt's folding rules; add / sub chains of constants are left out:
# they cancel, e.g. (x + a) + b -> x + (a + b) fails near x + a + b = 0)
(x * a) * b -> x * (a * b)   where a const, b const
(x / a) * b -> x * (b / a)   where a const, b const, a != 0
(a / x) * b -> (a * b) / x   where a const, b const
(x * a) / b -> x * (a / b)   where a const, b const, b != 0
a / (x / b) -> (a * b) / x   where a const, b const
a / (x * b) -> (a / b) / x   where a const, b const, b != 0
(x / a) / b -> x * (1.0 / (a * b))   where a const, b const, a != 0, b != 0
-(x * a) -> x * -a   where a const
(x >= e ? a : b) * k -> x >= e ? a * k : b * k   where a const, b const, k const
(x >= e ? a : b) + k -> x >= e ? a + k : b + k   where a const, b const, k const
lerp(a, b, 0.0) -> a
lerp(a, b, 1.0) -> b
dot(v, float3(1.0, 0.0, 0.0)) -> v.x   where v : float3

# factoring
a * b + a * c -> a * (b + c)
a * b - a * c -> a * (b - c)

# vectors
length(v) * length(v) -> dot(v, v)   where v : float3
normalize(v) * length(v) -> v        where v : float3
